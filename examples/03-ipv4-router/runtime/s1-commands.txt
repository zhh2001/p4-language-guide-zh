# 三张表的静态配置，只向尚未配置的新实例加载一次。
# run.sh 先过滤整行注释和空行，避免 CLI 报错或重复上一条命令。

# --- 路由表：目的 IP → (next hop, out port) ---
# 两台主机各有一条 /32 路由。不能把同网段其他地址也映射到这台主机。
table_add MyIngress.ipv4_lpm MyIngress.set_nhop 10.0.1.1/32 => 10.0.1.1 1
table_add MyIngress.ipv4_lpm MyIngress.set_nhop 10.0.2.2/32 => 10.0.2.2 2

# --- 静态邻居表：next hop IP → next hop MAC，不执行 ARP 协议 ---
table_add MyIngress.arp MyIngress.rewrite_dst_mac 10.0.1.1 => 00:00:00:00:01:01
table_add MyIngress.arp MyIngress.rewrite_dst_mac 10.0.2.2 => 00:00:00:00:02:02

# --- 出端口的源 MAC ---
table_add MyIngress.smac MyIngress.rewrite_src_mac 1 => 00:00:00:01:01:01
table_add MyIngress.smac MyIngress.rewrite_src_mac 2 => 00:00:00:02:02:02

table_dump MyIngress.ipv4_lpm
table_dump MyIngress.arp
table_dump MyIngress.smac
